theorem seed0_sym_run (g : Nat) {u : Nat} {ys : List Nat}
    (h : RNatRun seed0Forbidden (Live.mask seed0K) (Live.goal seed0K) u ys) :
    RNatRun seed0Forbidden (Live.mask seed0K) (Live.goal seed0K) u (ys.map (seed0SymWord g)) := by
  let a : Fin 8 := ⟨g % 8, Nat.mod_lt _ (by decide)⟩
  have hs := seed0_pair_checks a
  have heq : seed0SymWord g = Symmetry.permWord (151+151)
      (Symmetry.liftHalf 151 (seed0PairRow a)) := by
    funext word
    exact Symmetry.fastHalfPerm_eq 151 (seed0PairPerms.getD a []) hs.1 word
  have hp : ∀ j < 151, seed0PairRow a j < 151 := fun j hj => (hs.2 ⟨j,hj⟩).1
  have hq : ∀ j < 151, seed0PairRow (seed0InversePerms.getD a 0) j < 151 :=
    fun j hj => (hs.2 ⟨j,hj⟩).2.1
  have hinv : ∀ j < 151, seed0PairRow a (seed0PairRow (seed0InversePerms.getD a 0) j) = j :=
    fun j hj => (hs.2 ⟨j,hj⟩).2.2
  have hmod : seed0SymWord (g % 8) = seed0SymWord g := by
    funext word
    simp only [seed0SymWord, Nat.mod_mod]
  have hf : seed0Forbidden.Subset (seed0Forbidden.map (seed0SymWord g)) := by
    have hh := seed0_pool_checks a
    change seed0Forbidden.Subset (seed0Forbidden.map (seed0SymWord (g % 8))) at hh
    rw [hmod] at hh
    exact hh
  rw [heq] at hf
  have hh := Symmetry.permWord_liftHalf_run 151 (seed0PairRow a)
    (seed0PairRow (seed0InversePerms.getD a 0)) hp hq hinv hf h
  simpa only [heq, Live.mask, Live.goal, Live.lo, seed0K] using hh
