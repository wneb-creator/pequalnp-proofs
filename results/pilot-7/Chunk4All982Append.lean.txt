theorem digit_fin : ∀ a b : Fin 16, digit a.val b.val = a.val &&& b.val := by
  decide +kernel

theorem digit_eq (a b : Nat) (ha : a < 16) (hb : b < 16) :
    digit a b = a &&& b :=
  digit_fin ⟨a, ha⟩ ⟨b, hb⟩

def blockAnd : Nat → Nat → Nat → Nat
  | 0, _, _ => 0
  | fuel+1, a, b =>
    digit (a % 16) (b % 16) + 16 * blockAnd fuel (a/16) (b/16)

theorem blockAnd_eq_of_lt (fuel a b : Nat)
    (ha : a < 16^fuel) (hb : b < 16^fuel) :
    blockAnd fuel a b = a &&& b := by
  induction fuel generalizing a b with
  | zero =>
    have ha0 : a = 0 := by simpa using ha
    have hb0 : b = 0 := by simpa using hb
    subst a; subst b
    rfl
  | succ fuel ih =>
    have ha' : a / 16 < 16 ^ fuel := by
      apply (Nat.div_lt_iff_lt_mul (by decide : 0 < 16)).2
      simpa [Nat.pow_succ] using ha
    have hb' : b / 16 < 16 ^ fuel := by
      apply (Nat.div_lt_iff_lt_mul (by decide : 0 < 16)).2
      simpa [Nat.pow_succ] using hb
    rw [blockAnd, digit_eq _ _ (Nat.mod_lt _ (by decide)) (Nat.mod_lt _ (by decide)), ih _ _ ha' hb']
    have hmod := @Nat.and_mod_two_pow a b 4
    have hdiv := @Nat.and_div_two_pow a b 4
    norm_num only at hmod hdiv
    rw [← hmod, ← hdiv]
    exact Nat.mod_add_div _ _

def safeAnd (fuel a b : Nat) : Nat :=
  if a < 16^fuel ∧ b < 16^fuel then blockAnd fuel a b else a &&& b

theorem safeAnd_eq (fuel a b : Nat) : safeAnd fuel a b = a &&& b := by
  unfold safeAnd
  split
  next h => exact blockAnd_eq_of_lt fuel a b h.1 h.2
  next _ => rfl

theorem add_bitwise (a b : Nat) :
    a+b = (a ||| b)+(a &&& b) ∧ a+b = (a ^^^ b)+2*(a &&& b) := by
  induction b using Nat.div2Induction generalizing a with
  | ind b ih =>
    obtain rfl | hb := Nat.eq_zero_or_pos b
    · simp
    · have hh := ih hb (a := a/2)
      have aa := Nat.mod_add_div a 2
      have bb := Nat.mod_add_div b 2
      have oo := Nat.mod_add_div (a ||| b) 2
      have nn := Nat.mod_add_div (a &&& b) 2
      have xx := Nat.mod_add_div (a ^^^ b) 2
      have hm : (a &&& b) % 2 = (a%2) &&& (b%2) := Nat.and_mod_two_pow (n := 1)
      have hx : (a ^^^ b) % 2 = (a%2) ^^^ (b%2) := Nat.xor_mod_two_pow (n := 1)
      rw [Nat.or_mod_two, Nat.or_div_two] at oo
      rw [hm, Nat.and_div_two] at nn
      rw [hx, Nat.xor_div_two] at xx
      obtain ha | ha := Nat.mod_two_eq_zero_or_one a
      all_goals obtain hb | hb := Nat.mod_two_eq_zero_or_one b
      all_goals simp [ha,hb] at oo nn xx
      all_goals omega

def safeOr (fuel a b : Nat) : Nat := a + b - safeAnd fuel a b

def safeXor (fuel a b : Nat) : Nat := a + b - 2 * safeAnd fuel a b
