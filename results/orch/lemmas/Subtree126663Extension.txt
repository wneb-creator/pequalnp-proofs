theorem restrict_available {R S : Type} {u : Bool} (rows : S → R)
    {pool : List (Wire R)} {v : Wire R} (h : Available u pool v) :
    Available u (pool.map (restrictRows rows)) (restrictRows rows v) := by
  rcases h with h | ⟨b, rfl⟩ | ⟨hu, h⟩
  · exact Or.inl (List.mem_map.mpr ⟨v, h, rfl⟩)
  · exact Or.inr (Or.inl ⟨b, rfl⟩)
  · exact Or.inr (Or.inr ⟨hu, List.mem_map.mpr ⟨inv v, h, rfl⟩⟩)

theorem restrict_combine {R S : Type} (rows : S → R) (op : Bool) (a b : Wire R) :
    restrictRows rows (combine op a b) =
      combine op (restrictRows rows a) (restrictRows rows b) := by
  cases op <;> rfl

theorem restrict_reach {R S : Type} {u : Bool} (rows : S → R)
    {roots : List (Wire R)} {p bank} (h : Reach u roots p bank) :
    Reach u (roots.map (restrictRows rows)) p (bank.map (restrictRows rows)) := by
  induction h using Reach.ind with
  | start => exact Reach.start
  | two op h ha hb ih =>
    simpa only [List.map_cons, restrict_combine] using
      Reach.two op ih (restrict_available rows ha) (restrict_available rows hb)

theorem Slice_from_nine {u : Bool}
    (rows : NineSigned.Row → Row) (raw : List Sig) (T d : Sig)
    (roots9 : List NineSigned.Sig) (G9 : NineSigned.Sig) (L : Nat)
    (domain : ∀ r, bit (rows r) d = true)
    (goal : restrictRows rows (observe T) = NineSigned.observe G9)
    (covered : ∀ a ∈ raw, NineSigned.Available u (roots9.map NineSigned.observe)
      (restrictRows rows (observe a)))
    (properG : NineSigned.proper G9)
    (prior : NineSigned.Lower u roots9 G9 L) : Slice u raw T d L := by
  let f : Row → Row := fun r => rows (sliceNineCoarse r)
  let roots10 := roots9.map sliceNineLift
  have roots_lift : roots10.map observe =
      (roots9.map NineSigned.observe).map (restrictRows sliceNineCoarse) := by
    simp only [roots10, List.map_map, Function.comp_def, observe_sliceNineLift]
  have roots_section : (roots10.map observe).map (restrictRows sliceNineSection) =
      roots9.map NineSigned.observe := by
    simp only [roots10, List.map_map, Function.comp_def, restrict_sliceNineLift]
  have target_section : restrictRows sliceNineSection (restrictRows f (observe T)) =
      NineSigned.observe G9 := by
    calc
      _ = restrictRows rows (observe T) := by
        funext r
        simp only [restrictRows, f, sliceNineCoarse_section]
      _ = _ := goal
  refine ⟨f, roots10, fun r => domain _, ?_, ?_, ?_⟩
  · intro a ha
    rw [roots_lift]
    exact restrict_available sliceNineCoarse (covered a ha)
  · intro h
    obtain ⟨b, hb⟩ := h
    apply NineSigned.observe_proper properG
    refine ⟨b, ?_⟩
    rw [← target_section, hb]
    rfl
  · intro p bank hp hv
    have hp' := restrict_reach sliceNineSection hp
    rw [roots_section] at hp'
    have hv' := restrict_available sliceNineSection hv
    rw [target_section] at hv'
    exact prior p _ (nine_reach_of_reach hp') hv'

def successorOps (u : Bool) : List Nat := if u then [0, 1, 2, 3] else [0, 1]

lemma successorOps_complete {u : Bool} {d : Nat} (h : allowed u d) :
    d ∈ successorOps u := by
  cases u <;> simp [successorOps, allowed] at * <;> omega

def successorGuard (ws : List Sig) (cap : Nat) (a b : Sig) (d : Nat) (ka kb : Bool) : Bool :=
  decide (a ≠ b) && decide (bin d a b ∉ ws) &&
    decide (bin d a b ≠ zero) && decide (bin d a b ≠ one) &&
    decide ((childTwo ws a b d ka kb).length ≤ cap)

def successorLists (u : Bool) (ws : List Sig) (cap : Nat) : List (List Sig) :=
  ws.flatMap fun a => ws.flatMap fun b => (successorOps u).flatMap fun d =>
    [false, true].flatMap fun ka => [false, true].filterMap fun kb =>
      if successorGuard ws cap a b d ka kb then some (childTwo ws a b d ka kb) else none

lemma successorLists_complete {u ws cap a b d ka kb}
    (ha : a ∈ ws) (hb : b ∈ ws) (hd : allowed u d) (hab : a ≠ b)
    (fresh : bin d a b ∉ ws) (hp : proper (bin d a b))
    (sz : (childTwo ws a b d ka kb).length ≤ cap) :
    childTwo ws a b d ka kb ∈ successorLists u ws cap := by
  have guard : successorGuard ws cap a b d ka kb = true := by
    simp [successorGuard, hab, fresh, hp.1, hp.2, sz]
  unfold successorLists
  apply List.mem_flatMap.mpr
  refine ⟨a, ha, ?_⟩
  apply List.mem_flatMap.mpr
  refine ⟨b, hb, ?_⟩
  apply List.mem_flatMap.mpr
  refine ⟨d, successorOps_complete hd, ?_⟩
  apply List.mem_flatMap.mpr
  refine ⟨ka, by cases ka <;> simp, ?_⟩
  apply List.mem_filterMap.mpr
  exact ⟨kb, by cases kb <;> simp, by simp [guard]⟩

def rowSeparates (u : Bool) (r s : Row) (w : Sig) : Bool :=
  if u then bit r w != bit s w else bit r w && !(bit s w)

lemma live_rowSeparates {u T k S r s} (h : Live u T k S)
    (sep : rowSeparates u r s T = true) : ∃ w ∈ S, rowSeparates u r s w = true := by
  cases u with
  | true =>
    have diff : bit r T ≠ bit s T := by simpa [rowSeparates] using sep
    obtain ⟨w, hw, hd⟩ := live_row_difference h diff
    exact ⟨w, hw, by simpa [rowSeparates] using hd⟩
  | false =>
    have bits : bit r T = true ∧ bit s T = false := by simpa [rowSeparates] using sep
    obtain ⟨w, hw, yes, no⟩ := mono_row_difference h rfl bits.1 bits.2
    exact ⟨w, hw, by simp [rowSeparates, yes, no]⟩

def rowReject (u : Bool) (T : Sig) (ws : List Sig) (pair : Row × Row) : Bool :=
  rowSeparates u pair.1 pair.2 T && ws.all (fun w => !(rowSeparates u pair.1 pair.2 w))

lemma rowReject_sound {u T ws pair} (check : rowReject u T ws pair = true) :
    ∀ k, ¬ Live u T k ws.toFinset := by
  intro k run
  have both := Bool.and_eq_true_iff.mp check
  obtain ⟨w, hw, hsep⟩ := live_rowSeparates run both.1
  have hn := List.all_eq_true.mp both.2 w (List.mem_toFinset.mp hw)
  simp [hsep] at hn

def successorCheck (u : Bool) (T : Sig) (ws : List Sig) (cap : Nat)
    (known : List (Finset Sig)) (rows : List (Row × Row)) : Bool :=
  (successorLists u ws cap).all fun child =>
    decide (child.toFinset ∈ known) || rows.any (rowReject u T child)

lemma successorCheck_sound {u T ws cap known rows} (nd : ws.Nodup) (out : T ∉ ws)
    (knownSound : ∀ S ∈ known, Impossible u T (cap - 1) S)
    (check : successorCheck u T ws cap known rows = true) :
    Impossible u T cap ws.toFinset := by
  apply enum_step nd out
  intro a ha b hb d hd ka kb hab fresh hp sz
  have mem := successorLists_complete ha hb hd hab fresh hp sz
  have done := List.all_eq_true.mp check _ mem
  rcases Bool.or_eq_true_iff.mp done with hit | cut
  · exact knownSound _ (of_decide_eq_true hit)
  · obtain ⟨pair, hpair, good⟩ := List.any_eq_true.mp cut
    exact fun k _ => rowReject_sound good k

def cofactorCheck (u : Bool) (raw : List Sig) (d : Sig) (L cap : Nat)
    (ws : List Sig) (cs : List Choice) : Bool :=
  decide ws.Nodup && match makePool u d raw cs with
    | none => false
    | some pool => poolCut u d L cap cs.length pool ws

lemma cofactorCheck_sound {u raw T d L cap ws cs} (slice : Slice u raw T d L)
    (check : cofactorCheck u raw d L cap ws cs = true) :
    Impossible u T cap ws.toFinset := by
  have both := Bool.and_eq_true_iff.mp check
  have nd : ws.Nodup := of_decide_eq_true both.1
  cases built : makePool u d raw cs with
  | none => simp [built] at both
  | some pool =>
    have cut : poolCut u d L cap cs.length pool ws = true := by simpa [built] using both.2
    exact poolCut_sound slice cs built nd cut
