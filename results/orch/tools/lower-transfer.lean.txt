theorem available {R : Type} {bank : List (R → Bool)} {a : R → Bool}
    (h : C2444.Available bank a) : C1674.Available true bank a := by
  rcases h with h | h
  · exact Or.inl h
  · exact Or.inr (Or.inl h)

theorem available_inv {R : Type} {bank : List (R → Bool)} {a : R → Bool}
    (h : C2444.Available bank a) : C1674.Available true bank (C2444.inv a) := by
  rcases h with h | ⟨b,rfl⟩
  · refine Or.inr (Or.inr ⟨rfl,?_⟩)
    have hi : C1674.inv (C2444.inv a) = a := by
      funext r
      change Bool.not (Bool.not (a r)) = a r
      cases a r <;> rfl
    rw [hi]
    exact h
  · exact Or.inr (Or.inl ⟨!b,rfl⟩)

theorem reach {R : Type} {roots : List (R → Bool)} {p : Nat}
    {bank : List (R → Bool)} (h : C2444.Reach roots p bank) :
    C1674.Reach true roots p bank := by
  apply h (fun k ws => C1674.Reach true roots k ws)
  · rfl
  · intro k ws a hr ha
    refine ⟨true,ws,C2444.inv a,C2444.inv a,hr,
      available_inv ha,available_inv ha,?_⟩
    congr 1
    funext r
    simp only [C1674.combine,ite_true,Bool.or_self]
  · intro k ws a b op hr ha hb
    exact ⟨op,ws,a,b,hr,available ha,available hb,rfl⟩

theorem normalized_lower (roots : List C2444.Sig) (T : C2444.Sig) (L : Nat) :
    C1674.Lower true roots T L → C2444.Lower roots T L := by
  intro lower p bank hr ha
  exact lower p bank (reach hr) (available ha)

theorem vars_eq : C1674.vars = C2444.vars := rfl

theorem graphGoal_eq : C1674.graphGoal = C2444.graphGoal := rfl

theorem normalized_graph_lower (m L : Nat) :
    C1674.Lower true (C1674.vars m) (C1674.graphGoal m) L →
    C2444.Lower (C2444.vars m) (C2444.graphGoal m) L := by
  intro h
  apply normalized_lower
  simpa only [vars_eq,graphGoal_eq] using h
