open PEqualNP.CircuitLowerBounds

def termFamilyValue {n : ℕ} (F : Finset (Finset (Fin n))) (x : Fin n → Bool) : Bool :=
  decide (∃ A ∈ F, ∀ i ∈ A, x i = true)

def termFamilyCovers {n : ℕ} (F G : Finset (Finset (Fin n))) : Bool :=
  decide (∀ A ∈ F, ∃ B ∈ G, B ⊆ A)

def termFamilyEquiv {n : ℕ} (F G : Finset (Finset (Fin n))) : Bool :=
  termFamilyCovers F G && termFamilyCovers G F

def termFamilyProduct {n : ℕ} (F G : Finset (Finset (Fin n))) : Finset (Finset (Fin n)) :=
  (F ×ˢ G).image (fun p => p.1 ∪ p.2)

lemma termFamilyValue_eq_true {n : ℕ} (F : Finset (Finset (Fin n))) (x : Fin n → Bool) :
    termFamilyValue F x = true ↔ ∃ A ∈ F, ∀ i ∈ A, x i = true := by
  simp [termFamilyValue]

lemma termFamilyValue_input {n : ℕ} (i : Fin n) (x : Fin n → Bool) :
    termFamilyValue ({{i}} : Finset (Finset (Fin n))) x = x i := by
  simp [termFamilyValue]

lemma termFamilyValue_union {n : ℕ} (F G : Finset (Finset (Fin n))) (x : Fin n → Bool) :
    termFamilyValue (F ∪ G) x = true ↔
      termFamilyValue F x = true ∨ termFamilyValue G x = true := by
  simp only [termFamilyValue_eq_true, Finset.mem_union]
  constructor
  · rintro ⟨A, hA, h⟩
    rcases hA with hA | hA
    · exact Or.inl ⟨A, hA, h⟩
    · exact Or.inr ⟨A, hA, h⟩
  · rintro (⟨A, hA, h⟩ | ⟨A, hA, h⟩)
    · exact ⟨A, Or.inl hA, h⟩
    · exact ⟨A, Or.inr hA, h⟩

lemma termFamilyValue_product {n : ℕ} (F G : Finset (Finset (Fin n))) (x : Fin n → Bool) :
    termFamilyValue (termFamilyProduct F G) x = true ↔
      termFamilyValue F x = true ∧ termFamilyValue G x = true := by
  simp only [termFamilyValue_eq_true]
  constructor
  · rintro ⟨S, hS, h⟩
    obtain ⟨⟨A, B⟩, hAB, rfl⟩ := Finset.mem_image.mp hS
    obtain ⟨hA, hB⟩ := Finset.mem_product.mp hAB
    exact ⟨⟨A, hA, fun i hi => h i (Finset.mem_union_left B hi)⟩,
      ⟨B, hB, fun i hi => h i (Finset.mem_union_right A hi)⟩⟩
  · rintro ⟨⟨A, hA, ha⟩, ⟨B, hB, hb⟩⟩
    refine ⟨A ∪ B, Finset.mem_image.mpr ⟨(A, B), Finset.mem_product.mpr ⟨hA, hB⟩, rfl⟩, ?_⟩
    intro i hi
    rcases Finset.mem_union.mp hi with hi | hi
    · exact ha i hi
    · exact hb i hi

lemma termFamilyCovers_sound {n : ℕ} (F G : Finset (Finset (Fin n)))
    (hcov : termFamilyCovers F G = true) (x : Fin n → Bool) :
    termFamilyValue F x = true → termFamilyValue G x = true := by
  have hc : ∀ A ∈ F, ∃ B ∈ G, B ⊆ A := by
    simpa [termFamilyCovers] using hcov
  simp only [termFamilyValue_eq_true]
  rintro ⟨A, hA, ha⟩
  obtain ⟨B, hB, hBA⟩ := hc A hA
  exact ⟨B, hB, fun i hi => ha i (hBA hi)⟩

lemma termFamilyEquiv_sound {n : ℕ} (F G : Finset (Finset (Fin n)))
    (heq : termFamilyEquiv F G = true) (x : Fin n → Bool) :
    termFamilyValue F x = true ↔ termFamilyValue G x = true := by
  simp only [termFamilyEquiv, Bool.and_eq_true] at heq
  exact ⟨termFamilyCovers_sound F G heq.1 x, termFamilyCovers_sound G F heq.2 x⟩

lemma termFamilyValue_edges {n : ℕ} (E : Finset (Fin n × Fin n)) (x : Fin n → Bool) :
    termFamilyValue (E.image (fun e => ({e.1, e.2} : Finset (Fin n)))) x = true ↔
      ∃ e ∈ E, x e.1 = true ∧ x e.2 = true := by
  rw [termFamilyValue_eq_true]
  constructor
  · rintro ⟨A, hA, ha⟩
    obtain ⟨e, he, rfl⟩ := Finset.mem_image.mp hA
    exact ⟨e, he, ha e.1 (by simp), ha e.2 (by simp)⟩
  · rintro ⟨e, he, h1, h2⟩
    refine ⟨{e.1, e.2}, Finset.mem_image.mpr ⟨e, he, rfl⟩, ?_⟩
    intro i hi
    simp only [Finset.mem_insert, Finset.mem_singleton] at hi
    rcases hi with rfl | rfl
    · exact h1
    · exact h2

lemma termFamilyCertificate_wire
    (n : ℕ) (C : Circuit) (cert : ℕ → Finset (Finset (Fin n)))
    (hgate : ∀ (j : ℕ) (g : Gate), C.gates[j]? = some g →
      (g.op = GateOp.and ∨ g.op = GateOp.or) ∧
      ∃ a b : ℕ, g.inputs = [a, b] ∧ a ≠ b ∧ a < n + j ∧ b < n + j)
    (hinput : ∀ i : Fin n, cert i.val = {{i}})
    (hcert : ∀ (j : ℕ) (g : Gate) (a b : ℕ), C.gates[j]? = some g →
      g.inputs = [a, b] →
      (g.op = GateOp.and → termFamilyEquiv (cert (n + j))
        (termFamilyProduct (cert a) (cert b)) = true) ∧
      (g.op = GateOp.or → termFamilyEquiv (cert (n + j)) (cert a ∪ cert b) = true))
    (x : Fin n → Bool) (w : ℕ) (hw : w < n + C.size) :
    termFamilyValue (cert w) x = true ↔ (C.wires (List.ofFn x)).getD w false = true := by
  revert hw
  induction w using Nat.strong_induction_on with
  | ind w ih =>
    intro hw
    by_cases hi : w < n
    · have hv : (C.wires (List.ofFn x)).getD w false = x ⟨w, hi⟩ := by
        rw [wires_input _ _ _ (by simp; omega)]
        rw [List.getD_eq_getElem _ _ (by simp; omega)]
        simp
      rw [hinput ⟨w, hi⟩, termFamilyValue_input, hv]
    · have hj : w - n < C.gates.length := by unfold Circuit.size at hw; omega
      let g := C.gates[w - n]'hj
      have hg : C.gates[w - n]? = some g := by simp [g, hj]
      obtain ⟨hop, a, b, hab, _, ha, hb⟩ := hgate (w - n) g hg
      have hwe : n + (w - n) = w := by omega
      have hia := ih a (by omega) (by omega)
      have hib := ih b (by omega) (by omega)
      have hv := wire_value_gate C x (w - n) g a b hg hab ha hb
      rw [hwe] at hv
      rcases hop with hop | hop
      · have hc := termFamilyEquiv_sound _ _ ((hcert (w - n) g a b hg hab).1 hop) x
        rw [hwe] at hc
        calc
          termFamilyValue (cert w) x = true ↔
              termFamilyValue (termFamilyProduct (cert a) (cert b)) x = true := hc
          _ ↔ (C.wires (List.ofFn x)).getD a false = true ∧
              (C.wires (List.ofFn x)).getD b false = true := by
            rw [termFamilyValue_product, hia, hib]
          _ ↔ (C.wires (List.ofFn x)).getD w false = true := by
            rw [hv]
            simp [hop, GateOp.eval]
      · have hc := termFamilyEquiv_sound _ _ ((hcert (w - n) g a b hg hab).2 hop) x
        rw [hwe] at hc
        calc
          termFamilyValue (cert w) x = true ↔
              termFamilyValue (cert a ∪ cert b) x = true := hc
          _ ↔ (C.wires (List.ofFn x)).getD a false = true ∨
              (C.wires (List.ofFn x)).getD b false = true := by
            rw [termFamilyValue_union, hia, hib]
          _ ↔ (C.wires (List.ofFn x)).getD w false = true := by
            rw [hv]
            simp [hop, GateOp.eval]

lemma termFamilyCertificate_eval
    (n : ℕ) (C : Circuit) (cert : ℕ → Finset (Finset (Fin n)))
    (hgate : ∀ (j : ℕ) (g : Gate), C.gates[j]? = some g →
      (g.op = GateOp.and ∨ g.op = GateOp.or) ∧
      ∃ a b : ℕ, g.inputs = [a, b] ∧ a ≠ b ∧ a < n + j ∧ b < n + j)
    (hinput : ∀ i : Fin n, cert i.val = {{i}})
    (hcert : ∀ (j : ℕ) (g : Gate) (a b : ℕ), C.gates[j]? = some g →
      g.inputs = [a, b] →
      (g.op = GateOp.and → termFamilyEquiv (cert (n + j))
        (termFamilyProduct (cert a) (cert b)) = true) ∧
      (g.op = GateOp.or → termFamilyEquiv (cert (n + j)) (cert a ∪ cert b) = true))
    (ho : C.output < n + C.size) (x : Fin n → Bool) :
    C.eval (List.ofFn x) = true ↔ termFamilyValue (cert C.output) x = true :=
  (termFamilyCertificate_wire n C cert hgate hinput hcert x C.output ho).symm
