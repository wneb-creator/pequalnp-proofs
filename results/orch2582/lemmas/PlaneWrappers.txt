theorem plane_read_two_if_large (q N : ℕ) (I : Fin N → Fin N → Prop) [DecidableRel I]
    (C : Circuit) (hL : ∀ l, (Finset.univ.filter (fun p => I p l)).card = q + 1)
    (hsize : N * (q + 2) - 1 ≤ C.size) :
    ∃ D : Circuit, IsAO2 (N + N) D ∧
      (∀ y : Fin (N + N) → Bool, circuitFun (N + N) D y =
        decide (∃ p l : Fin N, I p l ∧ y (Fin.castAdd N p) = true ∧
          y (Fin.natAdd N l) = true)) ∧
      D.size ≤ C.size ∧ PEqualNP.ReadK.IsReadK (N + N) D 2 := by
  obtain ⟨D, hAO, hfun, hbound, hread⟩ := canonical_plane_read_one q N I hL
  exact ⟨D, hAO, hfun, hbound.trans hsize,
    ReadMono.isReadK_mono D (N + N) (by omega) hread⟩

theorem plane_clean_read_two (N : ℕ) (I : Fin N → Fin N → Prop) [DecidableRel I]
    (C : Circuit)
    (hpair : ∀ p p' : Fin N, p ≠ p' → ∃! l : Fin N, I p l ∧ I p' l)
    (hao : IsAO2 (N + N) C)
    (hfun : ∀ x : Fin (N + N) → Bool, circuitFun (N + N) C x =
      decide (∃ p l : Fin N, I p l ∧ x (Fin.castAdd N p) = true ∧
        x (Fin.natAdd N l) = true))
    (hclean : ∀ j < C.gates.length, (gateAt C j).op = GateOp.and →
      ∀ x : Fin (N + N) → Bool, gateVal (N + N) C j x = true →
        circuitFun (N + N) C x = true) :
    ∃ D : Circuit, IsAO2 (N + N) D ∧
      (∀ x : Fin (N + N) → Bool, circuitFun (N + N) D x =
        decide (∃ p l : Fin N, I p l ∧ x (Fin.castAdd N p) = true ∧
          x (Fin.natAdd N l) = true)) ∧
      D.size ≤ C.size ∧ PEqualNP.ReadK.IsReadK (N + N) D 2 := by
  classical
  let E : Finset (Fin N × Fin N) := Finset.univ.filter fun e => I e.1 e.2
  have hfour : ∀ (a a' b b' : Fin N), a ≠ a' → b ≠ b' →
      ¬ ((a, b) ∈ E ∧ (a, b') ∈ E ∧ (a', b) ∈ E ∧ (a', b') ∈ E) := by
    intro a a' b b' haa hbb he
    simp only [E, Finset.mem_filter, Finset.mem_univ, true_and] at he
    obtain ⟨c, hc, hu⟩ := hpair a a' haa
    exact hbb ((hu b ⟨he.1, he.2.2.1⟩).trans (hu b' ⟨he.2.1, he.2.2.2⟩).symm)
  have hcomp : ∀ x : Fin (N + N) → Bool, circuitFun (N + N) C x =
      decide (∃ (a b : Fin N), (a, b) ∈ E ∧
        x (Fin.castAdd N a) = true ∧ x (Fin.natAdd N b) = true) := by
    intro x
    simpa only [E, Finset.mem_filter, Finset.mem_univ, true_and] using hfun x
  obtain ⟨D, hAO, hDF, hsize, hread⟩ := clean_read_two_normalize N N E C hfour hao hcomp hclean
  refine ⟨D, hAO, ?_, hsize, hread⟩
  intro x
  simpa only [E, Finset.mem_filter, Finset.mem_univ, true_and] using hDF x
