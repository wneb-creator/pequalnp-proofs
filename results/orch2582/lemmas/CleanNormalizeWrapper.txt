lemma split_vertex_cover (p l : ℕ) (S : Finset (Fin (p + l))) :
    ∃ P : Finset (Fin p), ∃ L : Finset (Fin l),
      (∀ a, a ∈ P ↔ Fin.castAdd l a ∈ S) ∧
      (∀ b, b ∈ L ↔ Fin.natAdd p b ∈ S) ∧ P.card + L.card = S.card := by
  classical
  let P := Finset.univ.filter fun a : Fin p => Fin.castAdd l a ∈ S
  let L := Finset.univ.filter fun b : Fin l => Fin.natAdd p b ∈ S
  have hp (a : Fin p) : a ∈ P ↔ Fin.castAdd l a ∈ S := by simp [P]
  have hl (b : Fin l) : b ∈ L ↔ Fin.natAdd p b ∈ S := by simp [L]
  refine ⟨P, L, hp, hl, ?_⟩
  have hi : Function.Injective (Fin.castAdd l : Fin p → Fin (p + l)) := by
    intro a b hab
    have hv := congrArg Fin.val hab
    exact Fin.ext hv
  have hj : Function.Injective (Fin.natAdd p : Fin l → Fin (p + l)) := by
    intro a b hab
    apply Fin.ext
    have hv := congrArg Fin.val hab
    change p + a.val = p + b.val at hv
    omega
  have hd : Disjoint (P.image (Fin.castAdd l)) (L.image (Fin.natAdd p)) := by
    apply Finset.disjoint_left.mpr
    intro w hw hw'
    obtain ⟨a, ha, he⟩ := Finset.mem_image.mp hw
    obtain ⟨b, hb, he'⟩ := Finset.mem_image.mp hw'
    have hv := congrArg Fin.val (he.trans he'.symm)
    change a.val = p + b.val at hv
    have ha' := a.isLt
    omega
  have hs : S = P.image (Fin.castAdd l) ∪ L.image (Fin.natAdd p) := by
    ext w
    constructor
    · intro hw
      cases w using Fin.addCases with
      | left a => exact Finset.mem_union_left _ (Finset.mem_image.mpr ⟨a, (hp a).2 hw, rfl⟩)
      | right b => exact Finset.mem_union_right _ (Finset.mem_image.mpr ⟨b, (hl b).2 hw, rfl⟩)
    · intro hw
      rcases Finset.mem_union.mp hw with hw | hw
      · obtain ⟨a, ha, rfl⟩ := Finset.mem_image.mp hw
        exact (hp a).1 ha
      · obtain ⟨b, hb, rfl⟩ := Finset.mem_image.mp hw
        exact (hl b).1 hb
  rw [hs, Finset.card_union_of_disjoint hd,
    Finset.card_image_of_injective _ hi, Finset.card_image_of_injective _ hj]

theorem clean_read_two_normalize (p l : ℕ) (E : Finset (Fin p × Fin l)) (C : Circuit)
    (hfour : ∀ (a a' : Fin p) (b b' : Fin l), a ≠ a' → b ≠ b' →
      ¬ ((a, b) ∈ E ∧ (a, b') ∈ E ∧ (a', b) ∈ E ∧ (a', b') ∈ E))
    (hao : IsAO2 (p + l) C)
    (hcomp : ∀ x : Fin (p + l) → Bool, circuitFun (p + l) C x =
      decide (∃ (a : Fin p) (b : Fin l), (a, b) ∈ E ∧
        x (Fin.castAdd l a) = true ∧ x (Fin.natAdd p b) = true))
    (hclean : ∀ j < C.gates.length, (gateAt C j).op = GateOp.and →
      ∀ x : Fin (p + l) → Bool, gateVal (p + l) C j x = true →
        circuitFun (p + l) C x = true) :
    ∃ D : Circuit, IsAO2 (p + l) D ∧
      (∀ x : Fin (p + l) → Bool, circuitFun (p + l) D x =
        decide (∃ (a : Fin p) (b : Fin l), (a, b) ∈ E ∧
          x (Fin.castAdd l a) = true ∧ x (Fin.natAdd p b) = true)) ∧
      D.size ≤ C.size ∧ PEqualNP.ReadK.IsReadK (p + l) D 2 := by
  obtain ⟨S, hS, hbound⟩ := CleanReuse.clean_bound p l E C hfour hao hcomp hclean
  obtain ⟨P, L, hp, hl, hcard⟩ := split_vertex_cover p l S
  have hcover : ∀ e ∈ E, e.1 ∈ P ∨ e.2 ∈ L := by
    intro e he
    rcases hS e.1 e.2 he with ha | hb
    · exact Or.inl ((hp e.1).2 ha)
    · exact Or.inr ((hl e.2).2 hb)
  obtain ⟨D, hAO, hsize, hfun, hread⟩ := canonical_cover p l E P L hcover
  refine ⟨D, hAO, hfun, ?_, ReadMono.isReadK_mono D (p + l) (by omega) hread⟩
  omega
